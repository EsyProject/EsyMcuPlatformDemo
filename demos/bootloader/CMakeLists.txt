cmake_minimum_required(VERSION 3.24)
get_filename_component(DEMO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ESY_TARGET_DEVICE PICO2 CACHE STRING "Demo target: PICO2 or STM32F407ZGT6")
set(ESY_MCU_PLATFORM_DIR "${DEMO_ROOT}/external/EsyMcuPlatform" CACHE PATH "EsyMcuPlatform checkout")
include("${ESY_MCU_PLATFORM_DIR}/cmake/esy_target_device.cmake")
esy_configure_target_device()
if(ESY_TARGET_DEVICE STREQUAL "PICO2")
    set(ESY_BOARD_PATH "${DEMO_ROOT}/shared/pico2")
elseif(ESY_TARGET_DEVICE MATCHES "^STM32F407ZG(T6)?$")
    set(ESY_BOARD_PATH "${DEMO_ROOT}/shared/stm32f407")
else()
    message(FATAL_ERROR "This example supports PICO2 or STM32F407ZG/STM32F407ZGT6 only.")
endif()
set(ESY_FEATURE_IMPORT_FILE "${CMAKE_CURRENT_LIST_DIR}/esy_feature_import.cmake")
set(ESY_SPECS_DIR "")
set(ESY_RTOS_CONFIG_DIR "")
set(ESY_STM32_SUPPORT_ROOT "")
set(ESY_LINKER_SCRIPT_DEFAULT "${ESY_BOARD_PATH}/linker/bootloader.ld")
include("${ESY_MCU_PLATFORM_DIR}/cmake/esy_early_import.cmake")
if(ESY_TARGET_DEVICE STREQUAL "PICO2" AND NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
    message(FATAL_ERROR "The Pico 2 examples require the Arm RP2350 SDK platform.")
endif()
if(DEFINED ESY_APP_SLOT)
    message(FATAL_ERROR "The bootloader has a fixed address; do not set ESY_APP_SLOT.")
endif()
project(bootloader VERSION 1.0.0 LANGUAGES C CXX ASM)
add_subdirectory("${ESY_MCU_PLATFORM_DIR}" "${CMAKE_BINARY_DIR}/EsyMcuPlatform")
add_subdirectory("${DEMO_ROOT}/shared/security" "${CMAKE_BINARY_DIR}/demo_security")
set(ESY_FIRMWARE_TARGET bootloader)
add_executable(${ESY_FIRMWARE_TARGET} main.c)
target_link_libraries(${ESY_FIRMWARE_TARGET} PRIVATE
    esy_hal esy_bootloader demo_security esy_opt_default)
include("${ESY_BOARD_PATH}/firmware.cmake")
if(ESY_TARGET_DEVICE STREQUAL "PICO2")
    demo_pico_add_firmware(${ESY_FIRMWARE_TARGET} bootloader)
else()
    demo_stm32_add_startup(${ESY_FIRMWARE_TARGET} bootloader)
endif()
include("${ESY_MCU_PLATFORM_DIR}/cmake/esy_firmware.cmake")
