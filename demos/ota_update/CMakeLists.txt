cmake_minimum_required(VERSION 3.24)
get_filename_component(DEMO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ESY_TARGET_DEVICE PICO2 CACHE STRING "Demo target: PICO2 or STM32F407ZGT6")
set(ESY_MCU_PLATFORM_DIR "${DEMO_ROOT}/external/EsyMcuPlatform" CACHE PATH "EsyMcuPlatform checkout")
set(FREERTOS_KERNEL_PATH "${DEMO_ROOT}/external/FreeRTOS-Kernel" CACHE PATH "FreeRTOS checkout")
include("${ESY_MCU_PLATFORM_DIR}/cmake/esy_target_device.cmake")
esy_configure_target_device()
if(ESY_TARGET_DEVICE STREQUAL "PICO2")
    set(ESY_BOARD_PATH "${DEMO_ROOT}/shared/pico2")
    set(demo_device_config "${CMAKE_CURRENT_LIST_DIR}/config/pico2")
elseif(ESY_TARGET_DEVICE MATCHES "^STM32F407ZG(T6)?$")
    set(ESY_BOARD_PATH "${DEMO_ROOT}/shared/stm32f407")
    set(demo_device_config "${CMAKE_CURRENT_LIST_DIR}/config/stm32f407")
else()
    message(FATAL_ERROR "This example supports PICO2 or STM32F407ZG/STM32F407ZGT6 only.")
endif()
set(ESY_FEATURE_IMPORT_FILE "${CMAKE_CURRENT_LIST_DIR}/esy_feature_import.cmake")
set(ESY_SPECS_DIR "")
set(ESY_STM32_SUPPORT_ROOT "")
set(ESY_RTOS_CONFIG_DIR "${demo_device_config}")
set(ESY_DEMO_CONFIG_DIR "${ESY_RTOS_CONFIG_DIR}")
set(ESY_LINKER_SCRIPT_A "${ESY_BOARD_PATH}/linker/slot_a.ld")
set(ESY_LINKER_SCRIPT_B "${ESY_BOARD_PATH}/linker/slot_b.ld")
set(DEMO_APP_SLOT A CACHE STRING "Application slot (A or B)")
set_property(CACHE DEMO_APP_SLOT PROPERTY STRINGS A B)
set(ESY_APP_SLOT "${DEMO_APP_SLOT}")
set(DEMO_FIRMWARE_VERSION 1 CACHE STRING "Signed firmware version (positive uint32)")
if(NOT DEMO_FIRMWARE_VERSION MATCHES "^[1-9][0-9]*$" OR
   DEMO_FIRMWARE_VERSION GREATER 4294967295)
    message(FATAL_ERROR "DEMO_FIRMWARE_VERSION must be a positive decimal uint32.")
endif()
include("${ESY_MCU_PLATFORM_DIR}/cmake/esy_early_import.cmake")
if(ESY_TARGET_DEVICE STREQUAL "PICO2" AND NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
    message(FATAL_ERROR "The Pico 2 examples require the Arm RP2350 SDK platform.")
endif()
string(TOUPPER "${ESY_APP_SLOT}" ESY_APP_SLOT)
if(NOT ESY_APP_SLOT MATCHES "^[AB]$")
    message(FATAL_ERROR "ESY_APP_SLOT must be A or B.")
endif()
project(ota_update VERSION 1.0.0 LANGUAGES C CXX ASM)
add_subdirectory("${ESY_MCU_PLATFORM_DIR}" "${CMAKE_BINARY_DIR}/EsyMcuPlatform")
add_subdirectory("${DEMO_ROOT}/shared/security" "${CMAKE_BINARY_DIR}/demo_security")
set(ESY_FIRMWARE_TARGET ota_update)
add_executable(${ESY_FIRMWARE_TARGET} main.c)
target_include_directories(${ESY_FIRMWARE_TARGET} PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/config" "${demo_device_config}")
target_compile_definitions(${ESY_FIRMWARE_TARGET} PRIVATE "DEMO_FIRMWARE_VERSION=${DEMO_FIRMWARE_VERSION}UL")
target_link_libraries(${ESY_FIRMWARE_TARGET} PRIVATE
    esy_hal esy_os esy_ota_core esy_ota_flash esy_bootloader esy_common_utils demo_security esy_opt_default)
include("${ESY_BOARD_PATH}/firmware.cmake")
string(TOLOWER "${ESY_APP_SLOT}" demo_slot_lower)
if(ESY_TARGET_DEVICE STREQUAL "PICO2")
    demo_pico_add_firmware(${ESY_FIRMWARE_TARGET} "slot_${demo_slot_lower}")
else()
    demo_stm32_add_startup(${ESY_FIRMWARE_TARGET} "slot_${demo_slot_lower}")
endif()
include("${ESY_MCU_PLATFORM_DIR}/cmake/esy_firmware.cmake")
